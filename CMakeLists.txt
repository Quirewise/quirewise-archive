cmake_minimum_required(VERSION 3.21)
project(QuirewiseArchive VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(ZLIB REQUIRED)
find_package(LibLZMA REQUIRED)
file(GLOB_RECURSE QW_ARCHIVE_SOURCES CONFIGURE_DEPENDS src/*.c)
add_library(quirewise_archive STATIC ${QW_ARCHIVE_SOURCES})
add_library(Quirewise::Archive ALIAS quirewise_archive)
set_target_properties(quirewise_archive PROPERTIES EXPORT_NAME Archive POSITION_INDEPENDENT_CODE ON)
target_compile_features(quirewise_archive PUBLIC c_std_11)
target_include_directories(quirewise_archive PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>" PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src ${CMAKE_CURRENT_SOURCE_DIR}/src/core ${CMAKE_CURRENT_SOURCE_DIR}/src/formats/archive ${CMAKE_CURRENT_SOURCE_DIR}/src/formats/image ${CMAKE_CURRENT_SOURCE_DIR}/src/formats/page ${CMAKE_CURRENT_SOURCE_DIR}/src/formats/djvu ${CMAKE_CURRENT_SOURCE_DIR}/src/backends)
target_compile_definitions(quirewise_archive PRIVATE QUIREWISE_STANDALONE_ARCHIVE=1 QUIREWISE_STANDALONE_RAR=1 QUIREWISE_STANDALONE_SEVEN_ZIP=1)
target_link_libraries(quirewise_archive PRIVATE ZLIB::ZLIB LibLZMA::LibLZMA)
install(TARGETS quirewise_archive EXPORT QuirewiseArchiveTargets ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}" LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}" RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(DIRECTORY include/ DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
install(EXPORT QuirewiseArchiveTargets FILE QuirewiseArchiveTargets.cmake NAMESPACE Quirewise:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/QuirewiseArchive")
configure_package_config_file(cmake/QuirewiseArchiveConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/QuirewiseArchiveConfig.cmake" INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/QuirewiseArchive")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/QuirewiseArchiveConfigVersion.cmake" VERSION "${PROJECT_VERSION}" COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/QuirewiseArchiveConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/QuirewiseArchiveConfigVersion.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/QuirewiseArchive")
